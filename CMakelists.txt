cmake_minimum_required(VERSION 3.16)
project(musl_libc LANGUAGES C CXX)

add_library(musl_libc)



add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/bits
    COMMAND ${CMAKE_CURRENT_LIST_DIR}/tools/sed.exe ARGS -f ${CMAKE_CURRENT_LIST_DIR}/tools/mkalltypes.sed ${CMAKE_CURRENT_LIST_DIR}/arch/i386/bits/alltypes.h.in ${CMAKE_CURRENT_LIST_DIR}/include/alltypes.h.in  > ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h
    DEPENDS ${CMAKE_CURRENT_LIST_DIR}/tools/sed.exe ${CMAKE_CURRENT_LIST_DIR}/tools/mkalltypes.sed ${CMAKE_CURRENT_LIST_DIR}/arch/i386/bits/alltypes.h.in ${CMAKE_CURRENT_LIST_DIR}/include/alltypes.h.in)


add_custom_target(musl_libc_generated_files DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h)

add_dependencies(musl_libc musl_libc_generated_files)

target_include_directories(musl_libc
PUBLIC
    ${CMAKE_CURRENT_BINARY_DIR}
    arch/i386
    arch/generic
PRIVATE
    src/internal
    src/include
PUBLIC
    include
)


set(EXIT_SOURCES
    src/exit/abort.c
    #src/exit/abort_lock.c
    #src/exit/assert.c
    #src/exit/atexit.c
    src/exit/at_quick_exit.c
    #src/exit/exit.c
    src/exit/quick_exit.c
    src/exit/_Exit.c
)


set(THREAD_SOURCES
    src/thread/thrd_create.c
    src/thread/thrd_exit.c
    src/thread/win32_thread_impl.c
)

file(GLOB COMPLEX_SOURCES CONFIGURE_DEPENDS src/complex/*.c)
file(GLOB CTYPE_SOURCES CONFIGURE_DEPENDS src/ctype/*.c)
#file(GLOB EXIT_SOURCES CONFIGURE_DEPENDS src/exit/*.c)
file(GLOB MATH_SOURCES CONFIGURE_DEPENDS src/math/*.c)
file(GLOB MULTIBYTE_SOURCES CONFIGURE_DEPENDS src/multibyte/*.c)
file(GLOB PRNG_SOURCES CONFIGURE_DEPENDS src/prng/*.c)
file(GLOB REGEX_SOURCES CONFIGURE_DEPENDS src/regex/*.c)
file(GLOB SEARCH_SOURCES CONFIGURE_DEPENDS src/search/*.c)
file(GLOB STDLIB_SOURCES CONFIGURE_DEPENDS src/stdlib/*.c)
file(GLOB STDIO_SOURCES CONFIGURE_DEPENDS src/stdio/*.c)
file(GLOB STRING_SOURCES CONFIGURE_DEPENDS src/string/*.c)

target_sources(musl_libc
PRIVATE
    src/startup/win32/start.cpp
    ${COMPLEX_SOURCES}
    ${CTYPE_SOURCES}
    ${EXIT_SOURCES}
    ${MATH_SOURCES}
    ${MULTIBYTE_SOURCES}
    ${PRNG_SOURCES}
    ${REGEX_SOURCES}
    ${SEARCH_SOURCES}
    ${STDLIB_SOURCES}
    #${STDIO_SOURCES}
    ${STRING_SOURCES}
    ${THREAD_SOURCES}
)

target_compile_options(musl_libc PRIVATE /W0)
