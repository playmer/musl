cmake_minimum_required(VERSION 3.16)
project(libc LANGUAGES C CXX)

add_library(libc)


file(GLOB COMPLEX_SOURCES CONFIGURE_DEPENDS src/complex/*.c)
file(GLOB CTYPE_SOURCES CONFIGURE_DEPENDS src/ctype/*.c)
file(GLOB MATH_SOURCES CONFIGURE_DEPENDS src/math/*.c)
file(GLOB PRNG_SOURCES CONFIGURE_DEPENDS src/prng/*.c)
file(GLOB REGEX_SOURCES CONFIGURE_DEPENDS src/regex/*.c)
file(GLOB STDLIB_SOURCES CONFIGURE_DEPENDS src/stdlib/*.c)
file(GLOB STDIO_SOURCES CONFIGURE_DEPENDS src/stdio/*.c)
file(GLOB STRING_SOURCES CONFIGURE_DEPENDS src/string/*.c)

add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/bits
    COMMAND ${CMAKE_CURRENT_LIST_DIR}/tools/sed.exe ARGS -f ${CMAKE_CURRENT_LIST_DIR}/tools/mkalltypes.sed ${CMAKE_CURRENT_LIST_DIR}/arch/i386/bits/alltypes.h.in ${CMAKE_CURRENT_LIST_DIR}/include/alltypes.h.in  > ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h
    DEPENDS ${CMAKE_CURRENT_LIST_DIR}/tools/sed.exe ${CMAKE_CURRENT_LIST_DIR}/tools/mkalltypes.sed ${CMAKE_CURRENT_LIST_DIR}/arch/i386/bits/alltypes.h.in ${CMAKE_CURRENT_LIST_DIR}/include/alltypes.h.in)


add_custom_target(libc_generated_files DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/bits/alltypes.h)

add_dependencies(libc libc_generated_files)

target_include_directories(libc
PUBLIC
    ${CMAKE_CURRENT_BINARY_DIR}
    arch/i386
    arch/generic
PRIVATE
    src/internal
    src/include
PUBLIC
    include
)

target_sources(libc
PRIVATE
    src/startup/win32/start.cpp
    ${COMPLEX_SOURCES}
    ${CTYPE_SOURCES}
    ${MATH_SOURCES}
    ${PRNG_SOURCES}
    ${REGEX_SOURCES}
    ${STDLIB_SOURCES}
    ${STDIO_SOURCES}
    ${STRING_SOURCES}
)

target_compile_options(libc PRIVATE /W0)

add_executable(startup_main tests/startup_main.c)
target_link_libraries(startup_main PUBLIC libc)
